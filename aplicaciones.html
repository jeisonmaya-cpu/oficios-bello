ColCampoDescripción
AIDIdentificador único (timestamp)
BFechaFecha de la actuación (DD/MM/AAAA)
CHoraHora del registro (HH:MM)
DTipoAPLICACION | DEVOLUCION
ENombreNombre completo del contribuyente
FCédulaCédula o NIT
GExpedienteNo. expediente / radicado
HObligaciónTipo de obligación tributaria
IVigenciaVigencias involucradas
JValorRetenidoValor total embargado
KValorMovimientoValor a aplicar o devolver
LSaldoObligacionSaldo de la obligación tributaria
MSaldoDevolverSaldo calculado a devolver
NBancoEntidad financiera
ORefBancoNo. cuenta o referencia
PEstadoPENDIENTE | EN PROCESO | EJECUTADO | DEVUELTO | OBSERVACIÓN
QFuncionarioFuncionario que registra
RAreaÁrea o dependencia
SObservacionesNotas adicionales
📋 Código sugerido para el Apps Script
// Agregar en el doPost() del Apps Script existente:
case 'guardar_apdev':
  var sh = ss.getSheetByName(e.parameter.hoja || 'Aplicaciones_Devoluciones');
  if (!sh) sh = ss.insertSheet(e.parameter.hoja || 'Aplicaciones_Devoluciones');
  sh.appendRow([
    e.parameter.id, e.parameter.fecha, e.parameter.hora,
    e.parameter.tipo, e.parameter.nombre, e.parameter.cedula,
    e.parameter.expediente, e.parameter.obligacion, e.parameter.vigencia,
    e.parameter.valorRetenido, e.parameter.valorMovimiento,
    e.parameter.saldoObligacion, e.parameter.saldoDevolver,
    e.parameter.banco, e.parameter.refBanco, e.parameter.estado,
    e.parameter.funcionario, e.parameter.area, e.parameter.observaciones
  ]);
  return ok('Registro guardado');

case 'listar_apdev':
  var sh2 = ss.getSheetByName(e.parameter.hoja || 'Aplicaciones_Devoluciones');
  if (!sh2 || sh2.getLastRow() < 2) return ContentService
    .createTextOutput(JSON.stringify({status:'ok',data:[]}))
    .setMimeType(ContentService.MimeType.JSON);
  var vals = sh2.getRange(2, 1, sh2.getLastRow()-1, 19).getValues();
  return ContentService
    .createTextOutput(JSON.stringify({status:'ok',data:vals}))
    .setMimeType(ContentService.MimeType.JSON);

case 'actualizar_estado_apdev':
  // Busca por ID y actualiza columna Estado (P = col 16)
  var sh3 = ss.getSheetByName(e.parameter.hoja || 'Aplicaciones_Devoluciones');
  var data3 = sh3.getDataRange().getValues();
  for(var i=1;i<data3.length;i++){
    if(String(data3[i][0])===String(e.parameter.id)){
      sh3.getRange(i+1,16).setValue(e.parameter.estado);
      sh3.getRange(i+1,19).setValue(e.parameter.obs || data3[i][18]);
      break;
    }
  }
  return ok('Estado actualizado');
