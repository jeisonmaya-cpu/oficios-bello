case 'listar_recaudo':

case 'guardar_gestion_recaudo':
  // Busca por depósito judicial (col A) y actualiza columnas de gestión
  var sh2 = ss.getSheetByName(e.parameter.hoja || 'Recaudo_Historico');
  var datos = sh2.getDataRange().getValues();
  var deposito = String(e.parameter.deposito);
  for (var i = 1; i < datos.length; i++) {
    if (String(datos[i][0]) === deposito) {
      // Columnas de gestión (ajustar índices según su hoja):
      sh2.getRange(i+1, 17).setValue(e.parameter.proceso);      // Proceso
      sh2.getRange(i+1, 18).setValue(e.parameter.impuesto);     // Impuesto
      sh2.getRange(i+1, 20).setValue(e.parameter.convenio);     // Pendiente convenio
      sh2.getRange(i+1, 21).setValue(e.parameter.observacion);  // Observacion
      sh2.getRange(i+1, 22).setValue(e.parameter.telefono);     // Telefono
      sh2.getRange(i+1, 23).setValue(e.parameter.correo);       // Correo
      sh2.getRange(i+1, 24).setValue(e.parameter.direccion);    // Direccion
      sh2.getRange(i+1, 25).setValue(e.parameter.ciudad);       // Ciudad
      sh2.getRange(i+1, 27).setValue(e.parameter.responsable);  // Responsable
      sh2.getRange(i+1, 28).setValue(e.parameter.gestionTes);   // Gestion Tesoreria
      sh2.getRange(i+1, 29).setValue(e.parameter.ciInicial);    // CI Inicial
      sh2.getRange(i+1, 30).setValue(e.parameter.compEgreso);   // Comprobante Egreso
      sh2.getRange(i+1, 31).setValue(e.parameter.radicadoResp); // Radicado Respuesta
      sh2.getRange(i+1, 32).setValue(e.parameter.estDev);       // Estado Devolucion
      sh2.getRange(i+1, 33).setValue(e.parameter.estApl);       // Estado Aplicacion
      break;
    }
  }
  return json({status:'ok'});
